# M4 — Combined scoring + confidence + counter-argument + bias

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Take the publisher tier (M3) and the AI content score (M2) and combine them into a single overall credibility score per spec §6. Add a confidence indicator (high/medium/low). Add an LLM-estimated bias scale. Add a counter-argument paragraph generated by the same LLM call. Add a cheap-signals fallback so unknown publishers get a non-zero source-rep score derived from the article itself. Mobile UI surfaces the combined score prominently with the sub-scores in a drill-down.

**Architecture:**

- New backend service `scoring.ts` — pure functions for `computeSourceRep`, `computeBiasAdjustment`, `computeOverall`, `bucketLabel`, `deriveConfidence`. All deterministic, no LLM, no I/O, fully unit-tested.
- New backend service `cheap-signals.ts` — synchronous signals derivable from already-extracted article (HTTPS, byline, article length, siteName present). Returns a score 30–60 for unknown publishers. Real WHOIS + contact-page checks deferred to M6.
- `llm-analyzer.ts` extended — single Claude call now returns `{contentQuality, contentReasoning, biasScale, counterArgument}` (one prompt, one call, four fields). Cheaper than separate calls.
- `stub-analyzer.ts` matches new shape (fixed values for all four fields).
- `routes/check.ts` runs scoring after analysis: combines publisher tier + LLM result + cheap-signals (if needed) → combined `overall`, `bucket`, `confidence`, `bias`.
- Mobile result UI: combined score is the headline (big colored number + bucket label + confidence pill). Sub-scores in a drill-down section below.

**Tech stack:** no new libraries.

**Out of scope (deferred):**
- WHOIS / domain-age lookup → M6 (network call adds latency + dependency)
- Outbound-link count from raw HTML → M6 (requires passing raw HTML alongside Readability output)
- Full "disagreement transparency" UI (multiple sources rating bias differently) → M5 or once AllSides clears
- Deep Analysis multi-agent debate → M5
- Caching / rate-limiter / cost circuit-breaker / SSRF / Postgres → M6
- Maestro mobile E2E + LLM eval suite + feedback widget → M6

import type {
  Confidence,
  OverallBucket,
  ReliabilityTier
} from '../types/api.js';

const TIER_TO_SCORE: Record<Exclude<ReliabilityTier, 'unknown'>, number> = {
  high: 90,
  medium: 65,
  mixed: 50,
  low: 15
};

export function sourceRepFromTier(tier: ReliabilityTier): number | null {
  if (tier === 'unknown') return null;
  return TIER_TO_SCORE[tier];
}

/**
 * Bias adjustment: 100 - |bias| × 10, clamped to [50, 100].
 * Center (0) → 100 (no penalty). Extreme (±5) → 50 (max penalty).
 * Out-of-range inputs clamp at the ±5 extreme.
 */
export function biasAdjustment(bias: number): number {
  const clamped = Math.max(-5, Math.min(5, bias));
  return 100 - Math.abs(clamped) * 10;
}

export interface OverallInputs {
  sourceRep: number;       // 0-100
  contentQuality: number;  // 0-100
  bias: number;            // -5..+5
}

/**
 * Combined weighted score per spec §6:
 *   Overall = 0.45·SourceRep + 0.45·ContentQuality + 0.10·BiasAdjustment
 * Inputs clamped, output rounded to integer.
 */
export function computeOverall(inputs: OverallInputs): number {
  const sr = Math.max(0, Math.min(100, inputs.sourceRep));
  const cq = Math.max(0, Math.min(100, inputs.contentQuality));
  const ba = biasAdjustment(inputs.bias);
  return Math.round(0.45 * sr + 0.45 * cq + 0.10 * ba);
}

export function bucketLabel(overall: number): OverallBucket {
  if (overall >= 80) return 'highly-credible';
  if (overall >= 60) return 'mostly-credible';
  if (overall >= 40) return 'mixed';
  if (overall >= 20) return 'questionable';
  return 'low';
}

export interface ConfidenceInputs {
  datasetCount: number;      // number of dataset verdicts covering the publisher
  extractionOk: boolean;     // was the article body successfully extracted
}

/**
 * Spec §6 deterministic confidence:
 *   high   — 2+ datasets AND extraction ok
 *   medium — 1 dataset AND extraction ok, OR 2+ datasets without extraction
 *   low    — 0 datasets (cheap-signals fallback only)
 */
export function deriveConfidence(inputs: ConfidenceInputs): Confidence {
  if (inputs.datasetCount === 0) return 'low';
  if (inputs.datasetCount >= 2 && inputs.extractionOk) return 'high';
  return 'medium';
}

import type { CheapSignalsBreakdown } from '../types/api.js';

export interface CheapSignalsInput {
  url: string;
  byline?: string;
  siteName?: string;
  articleText: string;
}

const BASELINE = 30;
const HTTPS_BONUS = 5;
const BYLINE_BONUS = 10;
const SITENAME_BONUS = 5;
const LONG_BONUS = 10;
const LONG_THRESHOLD = 2000;

export function cheapSignals(input: CheapSignalsInput): CheapSignalsBreakdown {
  let url: URL;
  try {
    url = new URL(input.url);
  } catch {
    // Fall back: treat as http
    url = { protocol: 'http:' } as URL;
  }
  const https = url.protocol === 'https:';
  const hasByline = !!(input.byline && input.byline.trim().length > 0);
  const hasSiteName = !!(input.siteName && input.siteName.trim().length > 0);
  const articleLength = input.articleText.length;
  const longEnoughBonus = articleLength >= LONG_THRESHOLD;

  let score = BASELINE;
  if (https) score += HTTPS_BONUS;
  if (hasByline) score += BYLINE_BONUS;
  if (hasSiteName) score += SITENAME_BONUS;
  if (longEnoughBonus) score += LONG_BONUS;

  return { https, hasByline, hasSiteName, articleLength, longEnoughBonus, score };
}

import Anthropic from '@anthropic-ai/sdk';
import type { ApiError } from '../types/api.js';
import { CONTENT_ANALYSIS_SYSTEM_PROMPT, buildUserMessage } from '../prompts/content-analysis.js';

export interface AnalyzeOptions {
  client: Anthropic;
  model: string;
}

export interface AnalyzeOk {
  ok: true;
  contentQuality: number;
  contentReasoning: string;
  biasScale: number;
  counterArgument: string;
}

export type AnalyzeResult =
  | AnalyzeOk
  | { ok: false; error: ApiError['error'] };

interface ParsedLlmOutput {
  contentQuality: number;
  contentReasoning: string;
  biasScale: number;
  counterArgument: string;
}

function parseAndValidate(text: string): ParsedLlmOutput | null {
  const cleaned = text.trim().replace(/^```(?:json)?\s*|```\s*$/g, '');
  let obj: unknown;
  try {
    obj = JSON.parse(cleaned);
  } catch {
    return null;
  }
  if (typeof obj !== 'object' || obj === null) return null;
  const o = obj as Record<string, unknown>;
  const q = o.contentQuality;
  const r = o.contentReasoning;
  const b = o.biasScale;
  const ca = o.counterArgument;
  if (typeof q !== 'number' || !Number.isFinite(q) || q < 0 || q > 100) return null;
  if (typeof r !== 'string' || r.length === 0 || r.length > 2_000) return null;
  if (typeof b !== 'number' || !Number.isFinite(b) || b < -5 || b > 5) return null;
  if (typeof ca !== 'string' || ca.length === 0 || ca.length > 2_000) return null;
  return {
    contentQuality: Math.round(q),
    contentReasoning: r,
    biasScale: Math.round(b),
    counterArgument: ca
  };
}

async function callOnce(client: Anthropic, model: string, userMessage: string): Promise<string> {
  const response = await client.messages.create({
    model,
    max_tokens: 800,
    system: CONTENT_ANALYSIS_SYSTEM_PROMPT,
    messages: [{ role: 'user', content: userMessage }]
  });
  const block = response.content.find((b) => b.type === 'text');
  if (!block || block.type !== 'text') return '';
  return block.text;
}

export async function analyzeContent(
  articleText: string,
  opts: AnalyzeOptions
): Promise<AnalyzeResult> {
  const userMessage = buildUserMessage(articleText);

  for (let attempt = 1; attempt <= 2; attempt++) {
    let text: string;
    try {
      text = await callOnce(opts.client, opts.model, userMessage);
    } catch (e) {
      return {
        ok: false,
        error: {
          code: 'LLM_UNAVAILABLE',
          message: 'AI analysis service is unavailable. Try again.',
          retryable: true
        }
      };
    }
    const parsed = parseAndValidate(text);
    if (parsed) {
      return { ok: true, ...parsed };
    }
  }

  return {
    ok: false,
    error: {
      code: 'LLM_MALFORMED',
      message: 'AI returned an invalid response. Please try again.',
      retryable: true
    }
  };
}

function lookupPublisher(url: string): PublisherLookup {
  const domain = canonicalDomain(url);
  if (!domain) {
    return { canonicalDomain: null, reliabilityTier: 'unknown', verdicts: [] };
  }
  if (isKnownShortener(domain)) {
    return {
      canonicalDomain: domain,
      reliabilityTier: 'unknown',
      verdicts: [{
        source: 'curated-wikipedia',
        verdict: `URL is a known shortener (${domain}). Resolve the link and check the destination.`
      }]
    };
  }
  return lookup(domain);
}

async function runPipeline(jobId: string, url: string): Promise<void> {
  const publisher = lookupPublisher(url);

  const extraction = await extractArticle(url);
  if (!extraction.ok) {
    store.update(jobId, { status: 'failed', publisher, error: extraction.error });
    return;
  }

  store.update(jobId, {
    status: 'partial',
    publisher,
    article: {
      title: extraction.article.title,
      byline: extraction.article.byline,
      siteName: extraction.article.siteName
    }
  });

  const analysis = await runAnalysis(extraction.article.text);
  if (!analysis.ok) {
    store.update(jobId, { status: 'failed', error: analysis.error });
    return;
  }

  // Compute sourceRep: tier-derived if known publisher, else cheap-signals.
  const tierScore = sourceRepFromTier(publisher.reliabilityTier);
  let sourceRep: number;
  let sourceRepFrom: 'curated-wikipedia' | 'cheap-signals';
  let cheap: ReturnType<typeof cheapSignals> | undefined;
  if (tierScore !== null) {
    sourceRep = tierScore;
    sourceRepFrom = 'curated-wikipedia';
  } else {
    cheap = cheapSignals({
      url,
      byline: extraction.article.byline,
      siteName: extraction.article.siteName,
      articleText: extraction.article.text
    });
    sourceRep = cheap.score;
    sourceRepFrom = 'cheap-signals';
  }

  const overall = computeOverall({
    sourceRep,
    contentQuality: analysis.contentQuality,
    bias: analysis.biasScale
  });

  store.update(jobId, {
    status: 'complete',
    overall,
    bucket: bucketLabel(overall),
    confidence: deriveConfidence({
      datasetCount: publisher.verdicts.length,
      extractionOk: true
    }),
    sourceRep,
    sourceRepFrom,
    cheapSignals: cheap,
    contentQuality: analysis.contentQuality,
    contentReasoning: analysis.contentReasoning,
    bias: { scale: analysis.biasScale, source: 'llm' },
    counterArgument: analysis.counterArgument
  });
}

        <TextInput
          style={styles.input}

          autoCapitalize="none"
          autoCorrect={false}
          keyboardType="url"
          value={url}
          onChangeText={setUrl}
          editable={!isWorking}
        />

        <Pressable
          accessibilityRole="button"
          style={({ pressed }) => [
            styles.button,
            pressed && styles.buttonPressed,
            isWorking && styles.buttonDisabled
          ]}
          disabled={isWorking}
          onPress={onSubmit}
        >
          <Text style={styles.buttonText}>Check</Text>
        </Pressable>

        <View style={styles.resultBox}>
          {state.kind === 'idle' && <Text style={styles.muted}>Awaiting a URL.</Text>}

          {state.kind === 'submitting' && (
            <View style={styles.inline}>
              <ActivityIndicator />
              <Text style={styles.muted}>Submitting…</Text>
            </View>
          )}

          {state.kind === 'polling' && (
            <View>
              {polling?.publisher && <PublisherDetailBlock publisher={polling.publisher} />}
              {polling?.article?.title && (
                <View style={{ marginTop: 12 }}>
                  <Text style={styles.label}>Article</Text>
                  <Text style={styles.value}>{polling.article.title}</Text>
                </View>
              )}
              <View style={[styles.inline, { marginTop: 12 }]}>
                <ActivityIndicator />
                <Text style={styles.muted}>Analyzing…</Text>
              </View>
            </View>
          )}

          {completeJob && (
            <View>
              <HeadlineBlock job={completeJob} />
              {completeJob.article?.title && (
                <View style={{ marginTop: 16 }}>
                  <Text style={styles.label}>Article</Text>
                  <Text style={styles.value}>{completeJob.article.title}</Text>
                </View>
              )}
              <BreakdownBlock job={completeJob} />

              {completeJob.publisher && <PublisherDetailBlock publisher={completeJob.publisher} />}

              {completeJob.counterArgument && (
                <View style={styles.counterBox}>
                  <Text style={styles.counterTitle}>Counter-argument</Text>
                  <Text style={styles.counterText}>{completeJob.counterArgument}</Text>
                  <Text style={styles.counterFootnote}>
                    The strongest reason this rating could be wrong. Media-literacy tool.
                  </Text>
                </View>
              )}

              <Text style={[styles.disclaimer, { marginTop: 16 }]}>
                AI-generated. May contain errors. Publisher tier from cited sources.
              </Text>

              <View style={styles.linkRow}>
                <Pressable onPress={onOpenMethodology}><Text style={styles.link}>How this works</Text></Pressable>
                <Pressable onPress={onOpenLimitations}><Text style={styles.link}>Limitations</Text></Pressable>
              </View>

              <Pressable onPress={onReset} style={styles.resetLink}>
                <Text style={styles.resetText}>Check another URL</Text>
              </Pressable>
            </View>
          )}

          {failedJob && failedJob.error && (
            <View>
              {failedJob.publisher && <PublisherDetailBlock publisher={failedJob.publisher} />}
              <Text style={[styles.errorTitle, { marginTop: 12 }]}>Couldn't analyze this URL</Text>
              <Text style={styles.errorMessage}>{failedJob.error.message}</Text>
              <Pressable onPress={onReset} style={styles.resetLink}>
                <Text style={styles.resetText}>Try another URL</Text>
              </Pressable>
            </View>
          )}

          {state.kind === 'error' && (
            <View>
              <Text style={styles.errorTitle}>Error</Text>
              <Text style={styles.errorMessage}>{state.message}</Text>
              <Pressable onPress={onReset} style={styles.resetLink}>
                <Text style={styles.resetText}>Try again</Text>
              </Pressable>
            </View>
          )}
        </View>
      </ScrollView>
    </SafeAreaView>
  );
}
