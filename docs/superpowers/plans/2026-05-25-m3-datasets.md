# M3 — Dataset lookups + canonical domain

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Add publisher-reputation lookups alongside the M2 content-quality score. Backend reads the article URL, canonicalizes the domain via `tldts` (handles AMP/subdomain/mobile/shortener), looks the domain up in a hand-curated reliability CSV + the IFCN signatory list, and surfaces those verdicts in the result. Mobile drill-down shows each dataset's verdict with attribution. Combined weighted scoring stays deferred to M4.

**Architecture:** New backend service `canonical-domain.ts` extracts the registrable domain. New service `dataset-lookup.ts` loads two snapshots (curated CSV + IFCN CSV) into in-memory `Map<canonicalDomain, verdict>` at boot and serves O(1) lookups. Pipeline runs canonicalize + lookup in parallel with extraction (no extra latency). New `datasets` field on `JobResult` carries an array of `{source, verdict, attribution}`. Mobile result UI renders each verdict with the source name visible (per spec §9 "always attributed").

**Tech Stack additions:** `tldts` (backend). No new mobile libraries.

**Out of scope for M3 (deferred):**
- AllSides integration → skipped for V1 unless license clears (spec §8); fall back to LLM-estimated bias in M4 if not.
- Postgres migration → deferred. In-memory `JobStore` still wipes on backend restart; acceptable for personal test V1. Land Postgres at M6 alongside other production hardening.
- Combined weighted overall score (`Overall = 0.45·SourceRep + 0.45·ContentQuality + 0.10·BiasAdjustment`) → M4.
- Confidence indicator (high/medium/low) → M4.
- Counter-argument view → M4.
- Disagreement-transparency UI ("MBFC says X, AllSides says Y") → M4.
- Cheap-signals fallback for unknown sites → M4.
- Deep Analysis (multi-agent debate) → M5.
- Caching / rate-limiter / cost circuit-breaker / SSRF → M6.

```ts
import { parse } from 'tldts';

const KNOWN_SHORTENERS = new Set([
  't.co',
  'bit.ly',
  'goo.gl',
  'tinyurl.com',
  'ow.ly',
  'buff.ly',
  'lnkd.in',
  'is.gd',
  'tiny.cc',
  'short.io',
  'rebrand.ly'
]);

/**
 * Return the registrable domain for a URL, lowercased.
 * Strips www/m/amp/mobile subdomains via tldts public-suffix logic.
 * Returns null for invalid URLs or IP-host URLs.
 */
export function canonicalDomain(url: string): string | null {
  const result = parse(url);
  if (!result.domain) return null;
  if (result.isIp) return null;
  return result.domain.toLowerCase();
}

/**
 * Detect URL shorteners. Caller should resolve redirects before dataset lookup
 * when this returns true — `t.co/abc` is not a publisher, it's a redirect host.
 */
export function isKnownShortener(host: string): boolean {
  return KNOWN_SHORTENERS.has(host.toLowerCase());
}
```

```markdown
Wikipedia's Perennial Sources page is freeform prose, not a structured feed.
Auto-scraping breaks regularly when editors restructure sections. ~30
high-traffic publishers cover the vast majority of articles users paste,
and a slow expansion path stays safer.
```

```ts
import { readFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import path from 'node:path';
import type {
  DatasetVerdict,
  PublisherLookup,
  ReliabilityTier
} from '../types/api.js';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const DATA_DIR = path.resolve(__dirname, '..', 'data');

const TIER_TO_VERDICT: Record<Exclude<ReliabilityTier, 'unknown'>, string> = {
  high: 'Generally reliable',
  medium: 'No consensus',
  mixed: 'Mixed reliability — depends on topic',
  low: 'Generally unreliable'
};

function parseCsv(text: string): Record<string, string>[] {
  const lines = text.split(/\r?\n/).filter((l) => l.length > 0 && !l.startsWith('#'));
  if (lines.length === 0) return [];
  const headers = lines[0].split(',').map((h) => h.trim());
  return lines.slice(1).map((line) => {
    const cells = line.split(',');
    const row: Record<string, string> = {};
    headers.forEach((h, i) => {
      row[h] = (cells[i] ?? '').trim();
    });
    return row;
  });
}

interface ReliabilityRow {
  tier: ReliabilityTier;
  wikipediaSection: string;
}

function loadCurated(): Map<string, ReliabilityRow> {
  const text = readFileSync(path.join(DATA_DIR, 'curated-reliability.csv'), 'utf-8');
  const rows = parseCsv(text);
  const map = new Map<string, ReliabilityRow>();
  for (const row of rows) {
    const domain = row.domain?.toLowerCase();
    const tier = row.tier as ReliabilityTier;
    if (!domain || !tier) continue;
    if (!['high', 'medium', 'mixed', 'low'].includes(tier)) continue;
    map.set(domain, { tier, wikipediaSection: row.wikipedia_section ?? '' });
  }
  return map;
}

function loadIfcn(): Map<string, string> {
  const text = readFileSync(path.join(DATA_DIR, 'ifcn-signatories.csv'), 'utf-8');
  const rows = parseCsv(text);
  const map = new Map<string, string>();
  for (const row of rows) {
    const domain = row.domain?.toLowerCase();
    if (!domain) continue;
    map.set(domain, row.organization ?? domain);
  }
  return map;
}

export type DatasetLookup = (domain: string) => PublisherLookup;

export function createDatasetLookup(): DatasetLookup {
  const curated = loadCurated();
  const ifcn = loadIfcn();

  return (rawDomain: string): PublisherLookup => {
    const domain = rawDomain.toLowerCase();
    const verdicts: DatasetVerdict[] = [];

    const curatedRow = curated.get(domain);
    if (curatedRow) {
      verdicts.push({
        source: 'curated-wikipedia',
        verdict: TIER_TO_VERDICT[curatedRow.tier],
        details: curatedRow.wikipediaSection
          ? `Wikipedia editors' consensus`
          : undefined,
        link: curatedRow.wikipediaSection
          ? `https://en.wikipedia.org/wiki/Wikipedia:Reliable_sources/Perennial_sources#${curatedRow.wikipediaSection}`
          : undefined
      });
    }

    const ifcnOrg = ifcn.get(domain);
    if (ifcnOrg) {
      verdicts.push({
        source: 'ifcn',
        verdict: `${ifcnOrg} is an IFCN-signatory fact-checker`,
        link: 'https://ifcncodeofprinciples.poynter.org/signatories'
      });
    }

    const reliabilityTier: ReliabilityTier = curatedRow ? curatedRow.tier : 'unknown';

    return {
      canonicalDomain: domain,
      reliabilityTier,
      verdicts
    };
  };
}
```

```ts
import { Hono } from 'hono';
import Anthropic from '@anthropic-ai/sdk';
import { isValidUrl } from '../services/url-validator.js';
import { createJobStore, type JobStore } from '../services/job-store.js';
import { extractArticle } from '../services/article-extractor.js';
import { analyzeContent, type AnalyzeResult } from '../services/llm-analyzer.js';
import { analyzeContentStub } from '../services/stub-analyzer.js';
import { canonicalDomain, isKnownShortener } from '../services/canonical-domain.js';
import { createDatasetLookup, type DatasetLookup } from '../services/dataset-lookup.js';
import { env, hasRealLlm } from '../config/env.js';
import type { ApiError, PublisherLookup } from '../types/api.js';

let store: JobStore = createJobStore();
let lookup: DatasetLookup = createDatasetLookup();
let client: Anthropic | null = hasRealLlm()
  ? new Anthropic({ apiKey: env.anthropicApiKey })
  : null;

export function setTestDeps(deps: {
  store?: JobStore;
  client?: Anthropic | null;
  lookup?: DatasetLookup;
}): void {
  if (deps.store) store = deps.store;
  if (deps.client !== undefined) client = deps.client;
  if (deps.lookup) lookup = deps.lookup;
}

export const checkRoutes = new Hono();

async function runAnalysis(articleText: string): Promise<AnalyzeResult> {
  if (client) {
    return analyzeContent(articleText, { client, model: env.anthropicModel });
  }
  return analyzeContentStub(articleText);
}

function lookupPublisher(url: string): PublisherLookup {
  const domain = canonicalDomain(url);
  if (!domain) {
    return { canonicalDomain: null, reliabilityTier: 'unknown', verdicts: [] };
  }
  if (isKnownShortener(domain)) {
    // M3 doesn't yet resolve shortener redirects; flag and move on.
    return {
      canonicalDomain: domain,
      reliabilityTier: 'unknown',
      verdicts: [{
        source: 'curated-wikipedia',
        verdict: `URL is a known shortener (${domain}). Resolve the link and check the destination.`
      }]
    };
  }
  return lookup(domain);
}

async function runPipeline(jobId: string, url: string): Promise<void> {
  // Publisher lookup is synchronous and free; run before extraction.
  const publisher = lookupPublisher(url);

  const extraction = await extractArticle(url);
  if (!extraction.ok) {
    // Still attach publisher info to a failed job — useful for unknown sites.
    store.update(jobId, { status: 'failed', publisher, error: extraction.error });
    return;
  }

  store.update(jobId, {
    status: 'partial',
    publisher,
    article: {
      title: extraction.article.title,
      byline: extraction.article.byline,
      siteName: extraction.article.siteName
    }
  });

  const analysis = await runAnalysis(extraction.article.text);
  if (!analysis.ok) {
    store.update(jobId, { status: 'failed', error: analysis.error });
    return;
  }

  store.update(jobId, {
    status: 'complete',
    contentQuality: analysis.score,
    contentReasoning: analysis.reasoning
  });
}

checkRoutes.post('/check', async (c) => {
  const body = await c.req.json().catch(() => null);

  if (!body || typeof body.url !== 'string' || typeof body.deviceId !== 'string') {
    const err: ApiError = {
      error: {
        code: 'URL_INVALID',
        message: 'Request must include {url, deviceId}.',
        retryable: false
      }
    };
    return c.json(err, 400);
  }
  if (!isValidUrl(body.url)) {
    const err: ApiError = {
      error: {
        code: 'URL_INVALID',
        message: 'URL is not a valid http or https URL.',
        retryable: false
      }
    };
    return c.json(err, 400);
  }

  const job = store.create({ url: body.url });
  runPipeline(job.jobId, body.url).catch(() => {
    store.update(job.jobId, {
      status: 'failed',
      error: {
        code: 'INTERNAL_ERROR',
        message: 'Unexpected error during analysis.',
        retryable: true
      }
    });
  });

  return c.json({ jobId: job.jobId, status: 'pending' }, 202);
});

checkRoutes.get('/check/:jobId', (c) => {
  const jobId = c.req.param('jobId');
  const job = store.get(jobId);
  if (!job) {
    const err: ApiError = {
      error: {
        code: 'INTERNAL_ERROR',
        message: 'Unknown jobId.',
        retryable: false
      }
    };
    return c.json(err, 404);
  }
  return c.json(job, 200);
});
```

```tsx
  const [url, setUrl] = useState('');
  const [state, setState] = useState<ScreenState>({ kind: 'idle' });

  async function pollUntilDone(jobId: string): Promise<void> {
    for (let attempt = 0; attempt < POLL_MAX_ATTEMPTS; attempt++) {
      const result = await getJob(jobId);
      if (!result.ok) {
        setState({ kind: 'error', message: result.error.message });
        return;
      }
      if (result.value.status === 'complete' || result.value.status === 'failed') {
        setState({ kind: 'result', job: result.value });
        return;
      }
      setState({ kind: 'polling', jobId, job: result.value });
      await new Promise((r) => setTimeout(r, POLL_INTERVAL_MS));
    }
    setState({ kind: 'error', message: 'Timed out waiting for the result.' });
  }

  async function onSubmit() {
    if (!url.trim()) return;
    setState({ kind: 'submitting' });
    const deviceId = await getDeviceId();
    const submission = await submitCheck(url.trim(), deviceId);
    if (!submission.ok) {
      setState({ kind: 'error', message: submission.error.message });
      return;

    setState({ kind: 'polling', jobId: submission.value.jobId });
    await pollUntilDone(submission.value.jobId);
```
